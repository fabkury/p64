# The GIFs in assets/gifs are embedded in the app binary (memory-mapped flash) and
# listed in a generated table, gif_assets.cpp. Add a file to the folder and rebuild.
#
# ESP-IDF evaluates this file once in script mode just to collect requirements; that
# pass allows neither CONFIGURE_DEPENDS nor writing files, so generation happens only
# in the real configure pass.
if(CMAKE_SCRIPT_MODE_FILE)
    file(GLOB GIF_FILES "${CMAKE_CURRENT_LIST_DIR}/../assets/gifs/*.gif")
else()
    file(GLOB GIF_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../assets/gifs/*.gif")
endif()
list(SORT GIF_FILES)
list(LENGTH GIF_FILES GIF_COUNT)

set(GIF_TABLE_SRC "${CMAKE_CURRENT_BINARY_DIR}/gif_assets.cpp")
if(NOT CMAKE_SCRIPT_MODE_FILE)
set(GIF_TABLE_CONTENT "// Generated by main/CMakeLists.txt from assets/gifs/*.gif. Do not edit.\n")
string(APPEND GIF_TABLE_CONTENT "#include \"gif_assets.hpp\"\n\n")
set(GIF_TABLE_ENTRIES "")
foreach(gif ${GIF_FILES})
    get_filename_component(gif_name "${gif}" NAME)
    # Same mangling as ESP-IDF's data_file_embed_asm.cmake: the whole symbol string
    # through MAKE_C_IDENTIFIER.
    string(MAKE_C_IDENTIFIER "_binary_${gif_name}_start" sym_start)
    string(MAKE_C_IDENTIFIER "_binary_${gif_name}_end" sym_end)
    string(APPEND GIF_TABLE_CONTENT "extern \"C\" const uint8_t ${sym_start}[];\n")
    string(APPEND GIF_TABLE_CONTENT "extern \"C\" const uint8_t ${sym_end}[];\n")
    string(APPEND GIF_TABLE_ENTRIES
        "    {\"${gif_name}\", ${sym_start}, static_cast<size_t>(${sym_end} - ${sym_start})},\n")
endforeach()
string(APPEND GIF_TABLE_CONTENT "\nnamespace p64 {\n\n")
string(APPEND GIF_TABLE_CONTENT "const GifAsset kGifAssets[] = {\n${GIF_TABLE_ENTRIES}")
string(APPEND GIF_TABLE_CONTENT "    {nullptr, nullptr, 0},  // keeps the array non-empty when there are no GIFs\n};\n")
string(APPEND GIF_TABLE_CONTENT "const size_t kGifAssetCount = ${GIF_COUNT};\n\n}  // namespace p64\n")
file(WRITE "${GIF_TABLE_SRC}.tmp" "${GIF_TABLE_CONTENT}")
# Only touch the real file when the content changed, so unchanged assets do not rebuild.
execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${GIF_TABLE_SRC}.tmp" "${GIF_TABLE_SRC}")
endif()

idf_component_register(
    SRCS
        "main.cpp"
        "display.cpp"
        "button.cpp"
        "gif_player.cpp"
        "net/wifi.cpp"
        "net/clock.cpp"
        "net/makapix.cpp"
        "net/speedtest.cpp"
        "net/web.cpp"
        "scenes/gif_show.cpp"
        "${GIF_TABLE_SRC}"
    INCLUDE_DIRS
        "."
    PRIV_REQUIRES
        esphome__esp-hub75
        animatedgif
        driver
        esp_timer
        nvs_flash
        esp_wifi
        esp_netif
        esp_event
        lwip
        esp_http_client
        esp-tls
        mbedtls
        json
        esp_app_format
        esp_http_server
        espressif__mdns
    EMBED_FILES
        ${GIF_FILES}
)

message(STATUS "p64: embedding ${GIF_COUNT} GIF(s) from assets/gifs")
